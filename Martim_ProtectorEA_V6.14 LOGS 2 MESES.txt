bool HasSufficientMargin()
{
   // Calcular margen requerido nuevo grupo
   double margin_required = 0;
   
   // TO
   margin_required += OrderCalcMargin(OP_BUY, Symbol(), TO_Lot, Ask);
   
   // H1 potencial (peor caso)
   margin_required += OrderCalcMargin(OP_SELL, Symbol(), Hedge1_Lot, Bid);
   
   // H2 potencial (peor caso)
   margin_required += OrderCalcMargin(OP_BUY, Symbol(), Hedge2_Lot, Ask);
   
   double margin_free = AccountFreeMargin();
   double margin_level = AccountMargin() / AccountEquity() * 100;
   
   // CONDICIONES SEGURIDAD:
   
   // 1. Margen libre suficiente para grupo completo
   if(margin_free < margin_required)
   {
      Print("⚠️ Margen insuficiente: Libre $", margin_free, " < Requerido $", margin_required);
      return false;
   }
   
   // 2. Mantener margen level < umbral (ej: 60%)
   if(margin_level > 60.0)
   {
      Print("⚠️ Margin level alto: ", margin_level, "% (máx 60%)");
      return false;
   }
   
   // 3. Reservar % mínimo margen libre (ej: 30% equidad)
   double min_margin_reserve = AccountEquity() * 0.30;
   if(margin_free < min_margin_reserve)
   {
      Print("⚠️ Reserva margen insuficiente: $", margin_free, " < $", min_margin_reserve);
      return false;
   }
   
   return true;
}

mql4
void ManageH2Exits()
{
   // 1. Verificar si H2 activo
   if(!H2_Active || !HasOpenH2Positions())
      return;
   
   // 2. Calcular métricas actuales
   double to_pips = CalculateTOProfitPips();
   double h2_pips = CalculateH2ProfitPips();
   double net_pips = to_pips + h2_pips;
   
   // 3. EVALUACIÓN BREAKEVEN OBLIGATORIO (Directriz 8)
   if(to_pips >= H2_Breakeven_TO_Pips)
   {
      if(EvaluateBreakevenExit(net_pips))
         return;  // Ya cerrado
   }
   
   // 4. EVALUACIÓN PRIMERA GANANCIA (Directriz 7)
   if(to_pips >= H2_FirstProfit_TO_Pips && !H2_FirstProfitReached)
   {
      if(EvaluateFirstProfitExit(net_pips))
         return;  // Ya cerrado
      
      H2_FirstProfitReached = true;  // Marcar alcanzado
   }
   
   // 5. EVALUACIÓN CONTINUA (Directriz 7 + Principio Salida)
   if(H2_FirstProfitReached)
   {
      EvaluateContinuousExit();
   }
}

bool EvaluateBreakevenExit(double net_pips)
{
   // Directriz 8: Si TO llega a breakeven y reversión clara
   // O si neto <= breakeven → SALIR OBLIGATORIO
   
   if(net_pips <= 0.0)  // Neto en cero o pérdida
   {
      CloseH2Group("Breakeven Obligatorio - Neto: " + DoubleToStr(net_pips, 1) + "p");
      return true;
   }
   
   // Si TO en breakeven pero indicadores contradicen → SALIR
   if(DetectStrongReversalSignal())
   {
      CloseH2Group("Breakeven + Reversión Clara - Neto: " + DoubleToStr(net_pips, 1) + "p");
      return true;
   }
   
   return false;
}

   {

      Log("H2: Esperando ganancia neta >= " + DoubleToStr(H2_FirstProfit_Net_Pips, 1) + "p | Actual: " + DoubleToStr(net_pips, 1) + "p");
      return false;
   }
   
   // Hay ganancia neta → Evaluar indicadores
   if(IndicatorsSupportContinue())
   {
      Log("H2: Primera ganancia alcanzada | Neto: " + DoubleToStr(net_pips, 1) + "p | Indicadores APOYAN - CONTINUAR");
      return false;  // Continuar
   }
   else
   {
      CloseH2Group("Primera Ganancia + Indicadores CONTRADICEN - Neto: " + DoubleToStr(net_pips, 1) + "p");
      return true;  // Salir
   }
}

void EvaluateContinuousExit()
{

   int signal_type = DetectReversalSignal();
   
   switch(signal_type)
   {
      case 1:  // RSI EXTREMO + retroceso >=5pts
         CloseH2Group("RSI Extremo Reversión");
         break;
         
      case 2:  // HMA reversión aguda
         CloseH2Group("HMA Reversión Aguda");

      case 3:  // Ambos indicadores (nivel moderado)

   }
}

mql4
bool DetectRSI_ExtremeReversal()
{
   double rsi_current = iCustom(NULL, 0, "RSI2", 0, 0);
   int trend_dir = GetCurrentH2Direction();  // 1=alcista, -1=bajista
   
   if(trend_dir == 1)  // Tendencia ALCISTA (BUY/H2_BUY)
   {
      // Buscar RSI > 90 y retroceso >= 5pts
      if(rsi_current > H2_RSI_Extreme_High)
      {
         if(rsi_current > H2_RSI_Max)
         {
            H2_RSI_Max = rsi_current;  // Actualizar máximo
            H2_RSI_ExtremeTime = TimeCurrent();
         }
      }
      
      // Detectar retroceso
      if(H2_RSI_Max > H2_RSI_Extreme_High)  // Ya pasó extremo
      {
         double retrace = H2_RSI_Max - rsi_current;
         if(retrace >= H2_RSI_Retrace_Points)
         {
            Log("RSI_EXTREME_REVERSAL | Max: " + DoubleToStr(H2_RSI_Max, 1) + 
                " | Current: " + DoubleToStr(rsi_current, 1) + 
                " | Retrace: " + DoubleToStr(retrace, 1) + "pts");
            return true;
         }
      }
   }
   else if(trend_dir == -1)  // Tendencia BAJISTA (SELL/H2_SELL)
   {
      // Buscar RSI < 10 y retroceso >= 5pts
      if(rsi_current < H2_RSI_Extreme_Low)
      {
         if(rsi_current < H2_RSI_Min)
         {
            H2_RSI_Min = rsi_current;  // Actualizar mínimo
            H2_RSI_ExtremeTime = TimeCurrent();
         }
      }
      
      // Detectar retroceso
      if(H2_RSI_Min < H2_RSI_Extreme_Low)  // Ya pasó extremo
      {
         double retrace = rsi_current - H2_RSI_Min;
         if(retrace >= H2_RSI_Retrace_Points)
         {
            Log("RSI_EXTREME_REVERSAL | Min: " + DoubleToStr(H2_RSI_Min, 1) + 
                " | Current: " + DoubleToStr(rsi_current, 1) + 
                " | Retrace: " + DoubleToStr(retrace, 1) + "pts");
            return true;
         }
      }
   }
   
   return false;
}
2. HMA REVERSIÓN AGUDA (Severo - suficiente solo)
mql4
bool DetectHMA_AcuteReversal()
{
   // Obtener últimas 3 velas HMA
   double hma_val2 = iCustom(NULL, 0, "HMA_Color", HMA_Period, 0, 2);
   double hma_val1 = iCustom(NULL, 0, "HMA_Color", HMA_Period, 0, 1);
   double hma_val0 = iCustom(NULL, 0, "HMA_Color", HMA_Period, 0, 0);
   
   int hma_color0 = iCustom(NULL, 0, "HMA_Color", HMA_Period, 1, 0);  // Buffer color
   int hma_color1 = iCustom(NULL, 0, "HMA_Color", HMA_Period, 1, 1);
   
   // Verificar cambio de color
   if(hma_color0 == hma_color1)
      return false;  // No hay cambio color
   
   // Calcular ángulo usando función proporcionada
   double angle = CalculateAngle(hma_val2, hma_val1, hma_val0);
   
   // ¿Ángulo agudo? (< 60°)
   if(angle >= H2_HMA_Acute_Angle)
      return false;  // No es agudo
   
   // Calcular movimiento previo (últimas 5 velas antes cambio)
   double hma_val6 = iCustom(NULL, 0, "HMA_Color", HMA_Period, 0, 6);
   double prev_move = MathAbs(hma_val1 - hma_val6) / Point;  // En pips
   
   // Calcular retroceso actual
   double current_retrace = MathAbs(hma_val0 - hma_val1) / Point;  // En pips
   
   // Retroceso >= 30% del movimiento previo?
   double retrace_pct = (prev_move > 0) ? (current_retrace / prev_move * 100.0) : 0;
   
   if(retrace_pct < H2_HMA_Retrace_Pct)
      return false;  // Retroceso insuficiente
   
   // REVERSIÓN AGUDA CONFIRMADA
   Log("HMA_ACUTE_REVERSAL | Angle: " + DoubleToStr(angle, 0) + "° | " +
       "Retrace: " + DoubleToStr(retrace_pct, 1) + "% | " +
       "PrevMove: " + DoubleToStr(prev_move, 1) + "p");
   
   return true;
}

double CalculateAngle(double val2, double val1, double val0)
{
   if(val2 == EMPTY_VALUE || val1 == EMPTY_VALUE || val0 == EMPTY_VALUE)
      return 0;
   
   double slope = (val0 - val2) / 2.0;
   double angle = MathArctan(MathAbs(slope) * 10000) * (180.0 / M_PI);
   return MathRound(angle);
}

double CalculateTOProfitPips()
{
   // Calcular ganancia TO en pips
   double profit = 0;
   for(int i = OrdersTotal() - 1; i >= 0; i--)
   {
      if(!OrderSelect(i, SELECT_BY_POS)) continue;
      if(OrderMagicNumber() != MagicNumber) continue;
      if(OrderSymbol() != Symbol()) continue;
      if(OrderComment() != "TO") continue;
      
      profit += (OrderType() == OP_BUY) ? 
                (Bid - OrderOpenPrice()) / Point :
                (OrderOpenPrice() - Ask) / Point;
   }
   return profit;
}

double CalculateH2ProfitPips()
{
   // Similar para H2
   double profit = 0;
   for(int i = OrdersTotal() - 1; i >= 0; i--)
   {
      if(!OrderSelect(i, SELECT_BY_POS)) continue;
      if(OrderMagicNumber() != MagicNumber) continue;
      if(OrderSymbol() != Symbol()) continue;
      if(OrderComment() != "H2") continue;
      
      profit += (OrderType() == OP_BUY) ? 
                (Bid - OrderOpenPrice()) / Point :
                (OrderOpenPrice() - Ask) / Point;
   }
   return profit;
}

int GetCurrentH2Direction()
{
   // Obtener dirección H2 actual
   for(int i = OrdersTotal() - 1; i >= 0; i--)
   {
      if(!OrderSelect(i, SELECT_BY_POS)) continue;
      if(OrderMagicNumber() != MagicNumber) continue;
      if(OrderSymbol() != Symbol()) continue;
      if(OrderComment() != "H2") continue;
      
      return (OrderType() == OP_BUY) ? 1 : -1;
   }
   return 0;

void ManageH2Exits()
{
   if(!H2_Active || !HasOpenH2Positions())
      return;
   
   // 1. Estado actual
   double net_pips = CalculateGroupNetProfitPips();  // TO + H2
   double net_money = CalculateGroupNetProfitMoney();
   
   // 2. Determinar zona operación
   bool critical_zone = (net_pips <= H2_CriticalZone_Pips);
   
   // 3. Log estado
   if(critical_zone)
      Log("H2_ZONA_CRITICA | Net: " + DoubleToStr(net_pips, 1) + "p ($" + DoubleToStr(net_money, 2) + ")");
   
   // 4. Evaluar según zona
   if(critical_zone)
   {
      // CRITERIOS ESTRICTOS - Zona crítica
      EvaluateCriticalZone(net_pips, net_money);
   }
   else
   {
      // CRITERIOS NORMALES - Zona con ganancia
      EvaluateNormalZone(net_pips, net_money);
   }
}

void EvaluateCriticalZone(double net_pips, double net_money)
{
   // REGLA 1: PÉRDIDA → SALIR OBLIGATORIO
   if(net_money < 0)
   {
      CloseH2Group("ZONA_CRITICA - Pérdida | Net: $" + DoubleToStr(net_money, 2));
      return;
   }
   
   // REGLA 2: Ganancia mínima → Evaluar AMBOS indicadores
   // Solo continuar si AMBOS muy fuertes
   
   bool rsi_strong = IsRSI_Strong_Critical();
   bool hma_strong = IsHMA_Strong_Critical();
   
   if(!rsi_strong || !hma_strong)
   {
      // Alguno muestra debilidad → SALIR
      string reason = "ZONA_CRITICA - Debilidad | ";
      if(!rsi_strong) reason += "RSI_débil ";
      if(!hma_strong) reason += "HMA_débil ";
      reason += "| Net: " + DoubleToStr(net_pips, 1) + "p";
      
      CloseH2Group(reason);
      return;
   }
   
   // Ambos fuertes → Continuar con precaución
   Log("H2_CRITICA_CONTINUAR | Ambos fuertes | Net: " + DoubleToStr(net_pips, 1) + "p");
}

bool IsRSI_Strong_Critical()
{
   double rsi = iCustom(NULL, 0, "RSI2", 0, 0);
   int trend_dir = GetCurrentH2Direction();
   
   if(trend_dir == 1)  // ALCISTA
   {
      // RSI debe estar saludable (no extremo, no débil)
      if(rsi > H2_RSI_Weakness_High)  // >80 = debilidad
      {
         Log("RSI_CRITICA_DEBIL | RSI: " + DoubleToStr(rsi, 1) + " > " + H2_RSI_Weakness_High);
         return false;
      }
      
      // Además no debe estar retrocediendo significativamente
      if(H2_RSI_Max > 0)
      {
         double retrace = H2_RSI_Max - rsi;
         if(retrace > 3)  // Más estricto: >3pts ya es señal
         {
            Log("RSI_CRITICA_RETROCESO | Retrace: " + DoubleToStr(retrace, 1) + "pts");
            return false;
         }
      }
      
      return true;  // RSI fuerte
   }
   else if(trend_dir == -1)  // BAJISTA
   {
      if(rsi < H2_RSI_Weakness_Low)  // <20 = debilidad
      {
         Log("RSI_CRITICA_DEBIL | RSI: " + DoubleToStr(rsi, 1) + " < " + H2_RSI_Weakness_Low);
         return false;
      }
      
      if(H2_RSI_Min < 100)
      {
         double retrace = rsi - H2_RSI_Min;
         if(retrace > 3)
         {
            Log("RSI_CRITICA_RETROCESO | Retrace: " + DoubleToStr(retrace, 1) + "pts");
            return false;
         }
      }
      
      return true;
   }
   
   return false;
}

bool IsHMA_Strong_Critical()
{
   int hma_color_current = iCustom(NULL, 0, "HMA_Color", HMA_Period, 1, 0);
   int hma_color_prev = iCustom(NULL, 0, "HMA_Color", HMA_Period, 1, 1);
   int trend_dir = GetCurrentH2Direction();
   
   // 1. Color debe coincidir con tendencia
   if(trend_dir == 1 && hma_color_current != 1)  // Debe ser verde
   {
      Log("HMA_CRITICA_COLOR | Tendencia alcista pero HMA no verde");
      return false;
   }
   if(trend_dir == -1 && hma_color_current != 0)  // Debe ser rojo
   {
      Log("HMA_CRITICA_COLOR | Tendencia bajista pero HMA no rojo");
      return false;
   }
   
   // 2. NO debe haber cambiado color recientemente
   if(hma_color_current != hma_color_prev)
   {
      Log("HMA_CRITICA_CAMBIO | HMA cambió color recientemente");
      return false;
   }
   
   // 3. NO debe haber retroceso significativo
   double hma_val0 = iCustom(NULL, 0, "HMA_Color", HMA_Period, 0, 0);
   double hma_val1 = iCustom(NULL, 0, "HMA_Color", HMA_Period, 0, 1);
   
   double current_move_pips = MathAbs(hma_val0 - hma_val1) / Point;
   
   // Si la vela anterior tuvo movimiento
   if(H2_HMA_LastMove > 0)
   {
      double retrace_pct = (current_move_pips / H2_HMA_LastMove) * 100.0;
      
      // Si retrocede >20% (más estricto en zona crítica)
      if(retrace_pct > 20.0)
      {
         Log("HMA_CRITICA_RETROCESO | Retrace: " + DoubleToStr(retrace_pct, 1) + "%");
         return false;
      }
   }
   
   return true;  // HMA fuerte
}

            return true;
         }
      }
   }
   else if(trend_dir == -1)  // BAJISTA

HMA Reversión Aguda (CORREGIDO - 1 vela M5, no 5 velas)
mql4
bool DetectHMA_AcuteReversal()
{
   double hma_val2 = iCustom(NULL, 0, "HMA_Color", HMA_Period, 0, 2);
   double hma_val1 = iCustom(NULL, 0, "HMA_Color", HMA_Period, 0, 1);
   double hma_val0 = iCustom(NULL, 0, "HMA_Color", HMA_Period, 0, 0);
   
   int hma_color0 = iCustom(NULL, 0, "HMA_Color", HMA_Period, 1, 0);
   int hma_color1 = iCustom(NULL, 0, "HMA_Color", HMA_Period, 1, 1);
   
   // Debe haber cambio de color
   if(hma_color0 == hma_color1)
      return false;
   
   // Calcular ángulo
   double angle = CalculateAngle(hma_val2, hma_val1, hma_val0);
   if(angle >= H2_HMA_Acute_Angle)  // >=60° no es agudo
      return false;
   
   // Calcular retroceso respecto a VELA ANTERIOR (1 vela M5)
   double prev_move_pips = MathAbs(hma_val1 - hma_val2) / Point;
   double curr_retrace_pips = MathAbs(hma_val0 - hma_val1) / Point;
   
   if(prev_move_pips == 0)
      return false;
   
   double retrace_pct = (curr_retrace_pips / prev_move_pips) * 100.0;
   
   // Retroceso >= 35% de vela anterior
   if(retrace_pct >= H2_HMA_Retrace_Min_Pct)
   {
      Log("HMA_ACUTE | Angle: " + DoubleToStr(angle, 0) + "° | " +
          "Retrace: " + DoubleToStr(retrace_pct, 1) + "% (1 vela)");
      return true;
   }
   
   // O si es 2da vela consecutiva retrocediendo
   // y alcanza 80-100% de la primera vela
   double hma_val3 = iCustom(NULL, 0, "HMA_Color", HMA_Period, 0, 3);
   double total_retrace = MathAbs(hma_val0 - hma_val2) / Point;
   double first_move = MathAbs(hma_val2 - hma_val3) / Point;
   
   if(first_move > 0)
   {
      double total_pct = (total_retrace / first_move) * 100.0;
      if(total_pct >= H2_HMA_Retrace_2nd_Pct)  // >=80%
      {
         Log("HMA_ACUTE_2ND | Angle: " + DoubleToStr(angle, 0) + "° | " +
             "Retrace 2 velas: " + DoubleToStr(total_pct, 1) + "%");
         return true;
      }
   }
   
   return false;
}

•	Se generen nuevas señales casi inmediatamente

•	Control absoluto de frecuencia de trades

   if(Last_Signal_Time > 0 && hours_since_last < Min_Hours_Between_Signals)
      return;
   
   if(GroupCount > 0) return;
   if(!HasSufficientMargin()) return;
   
   // LEER INDICADORES
   double rsi_1 = iRSI(Symbol(), 0, RSI_Period, PRICE_CLOSE, 1);
   double rsi_2 = iRSI(Symbol(), 0, RSI_Period, PRICE_CLOSE, 2);

   int hma_color_2 = GetHMAColor(2);
   
   // FASE 1: DETECTAR SEÑAL POTENCIAL (barra N, shift=2)
   if(Pending_Signal_Type < 0)  // No hay señal pendiente
   {
      // Señal BUY potencial
      if(hma_color_2 == 1 && rsi_2 < RSI_Lower_Level)
      {
         Pending_Signal_Time = iTime(Symbol(), 0, 2);
         Pending_Signal_Type = OP_BUY;
         Pending_RSI_Value = rsi_2;
         Pending_HMA_Color = hma_color_2;
         
         Print("🔔 SEÑAL BUY PENDIENTE:");
         Print("   RSI(2): ", DoubleToString(rsi_2, 2));
         Print("   Esperando confirmación vela siguiente...");
         return;
      }
      
      // Señal SELL potencial
      if(hma_color_2 == 0 && rsi_2 > RSI_Upper_Level)
      {
         Pending_Signal_Time = iTime(Symbol(), 0, 2);
         Pending_Signal_Type = OP_SELL;
         Pending_RSI_Value = rsi_2;
         Pending_HMA_Color = hma_color_2;
         
         Print("🔔 SEÑAL SELL PENDIENTE:");
         Print("   RSI(2): ", DoubleToString(rsi_2, 2));
         Print("   Esperando confirmación vela siguiente...");
         return;
      }
   }
   
   // FASE 2: CONFIRMAR SEÑAL PENDIENTE (barra N+1, shift=1)
   if(Pending_Signal_Type >= 0)
   {
      // Verificar que pasó 1 barra
      datetime current_bar_time = iTime(Symbol(), 0, 1);
      if(current_bar_time <= Pending_Signal_Time)
         return;  // Aún no cerró nueva barra
      
      double close_1 = iClose(Symbol(), 0, 1);
      double open_1 = iOpen(Symbol(), 0, 1);
      
      // CONFIRMAR BUY
      if(Pending_Signal_Type == OP_BUY)
      {
         bool vela_alcista = (close_1 > open_1);
         bool rsi_subiendo = (rsi_1 > rsi_2);
         bool hma_mantiene = (hma_color_1 == 1);
         
         if(vela_alcista && rsi_subiendo && hma_mantiene)
         {
            Print("━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━");
            Print("✅ CONFIRMACIÓN BUY EXITOSA:");

            Print("   RSI(1): ", DoubleToString(rsi_1, 2), " (subiendo ✓)");
            Print("   HMA: ALCISTA ✓");
            Print("━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━");

            return;
         }

         bool rsi_bajando = (rsi_1 < rsi_2);
         bool hma_mantiene = (hma_color_1 == 0);
         
         if(vela_bajista && rsi_bajando && hma_mantiene)
